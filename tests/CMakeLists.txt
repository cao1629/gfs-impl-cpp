function(gfs_component_tests name lib)
  file(GLOB SRCS CONFIGURE_DEPENDS unit/${name}/*.cc)
  if(SRCS)
    add_executable(gfs_${name}_tests ${SRCS})
    target_link_libraries(gfs_${name}_tests PRIVATE ${lib} GTest::gtest GTest::gtest_main)
    add_test(NAME ${name} COMMAND gfs_${name}_tests)
  endif()
endfunction()

gfs_component_tests(common gfs_common)
gfs_component_tests(master gfs_master_lib)
gfs_component_tests(chunkserver gfs_chunkserver_lib)
gfs_component_tests(client gfs_client)

file(GLOB GFS_CLUSTER_TEST_SRCS CONFIGURE_DEPENDS integration/*.cc)
if(GFS_CLUSTER_TEST_SRCS)
  add_executable(gfs_cluster_tests ${GFS_CLUSTER_TEST_SRCS})
  target_link_libraries(gfs_cluster_tests PRIVATE gfs_client GTest::gtest GTest::gtest_main)
  target_compile_definitions(gfs_cluster_tests PRIVATE GFS_BIN_DIR="${CMAKE_BINARY_DIR}/bin")
  add_dependencies(gfs_cluster_tests gfs_master gfs_chunkserver)
  add_test(NAME cluster COMMAND gfs_cluster_tests)
endif()
